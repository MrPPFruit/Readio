# Readio M0 Technical Spikes Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a minimal Android spike app that proves Readio’s M0 assumptions: EPUB integration is viable, TXT can enter a unified reading model with fallback, PDF can stay a bounded compatibility path, and shared locator primitives are strong enough to support later progress/AI work.

**Architecture:** Start with a single Android `app` module that contains a debug-only spike harness with separate probes for EPUB, TXT, and PDF. Keep domain primitives in `core/model` so every spike uses the same `BookRef`, `Locator`, `Selection`, and `EvidenceAnchor` contracts. M0 ends with measured ADRs and exit criteria, not product polish.

**Tech Stack:** Kotlin, Jetpack Compose, AndroidX Navigation, coroutines, JUnit 4, Android instrumented tests, Readium Kotlin Toolkit (EPUB), Android `PdfRenderer` or AndroidX PDF (PDF), SAF/content URI.

---

## Scope Split Note

The spec covers four distinct phases: M0, M1, M2, and M3. This plan intentionally covers **M0 only**.

Do **not** mix M1 product work (full bookshelf/reader), M2 AI product work, or M3 search work into this plan. Those should become separate plans after M0 closes its technical unknowns.

This spike validates **fixture-based runtime viability only**. It does **not** validate SAF / `content://` ingestion, persisted URI permissions, or broken-source recovery paths; those remain M1 ingestion work.

```kotlin
package io.readio.core.io

import android.content.Context
import java.io.File

class AssetCopier(private val context: Context) {
    fun copyToCache(assetPath: String): File {
        val outFile = File(context.cacheDir, assetPath.substringAfterLast('/'))
        context.assets.open(assetPath).use { input ->
            outFile.outputStream().use { output -> input.copyTo(output) }
        }
        return outFile
    }
}
```

```kotlin
package io.readio.feature.txt

class TxtChapterParser {
    private val headingRegex = Regex("^(第.+[章回节卷].*|Chapter\\s+\\d+.*)$")

    fun parse(raw: String): List<TxtChapter> {
        val lines = raw.lines()
        val chapters = mutableListOf<TxtChapter>()
        var currentTitle = "全文"
        val currentBody = mutableListOf<String>()

        fun flush() {
            val body = currentBody.joinToString("\n").trim()
            if (currentBody.isEmpty() && currentTitle == "全文" && chapters.isEmpty()) {
                return
            }
            if (body.isNotBlank() || currentTitle == "全文") {
                chapters += TxtChapter(currentTitle, body)
            }
            currentBody.clear()
        }

        lines.forEach { line ->
            if (headingRegex.matches(line.trim())) {
                flush()
                currentTitle = line.trim()
            } else {
                currentBody += line
            }
        }
        flush()
        return chapters
    }
}
```

```kotlin
package io.readio.feature.epub

import android.content.Context
import io.readio.core.io.AssetCopier
import io.readio.core.model.BookFormat
import io.readio.core.model.Locator
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import org.readium.r2.shared.publication.Locator as ReadiumLocator
import org.readium.r2.shared.util.http.DefaultHttpClient
import org.readium.r2.shared.util.toUrl
import org.readium.r2.shared.util.getOrElse
import org.readium.r2.streamer.PublicationOpener
import org.readium.r2.streamer.parser.DefaultPublicationParser
import org.readium.r2.shared.util.asset.AssetRetriever

class ReadiumEpubEngine(
    private val context: Context,
    private val assetCopier: AssetCopier = AssetCopier(context),
) : EpubEngine {
    override suspend fun open(assetPath: String): EpubProbeResult = withContext(Dispatchers.IO) {
        val publication = openPublication(assetPath)
        val firstLink = publication.readingOrder.first()
        val firstReadiumLocator = ReadiumLocator(
            href = firstLink.href,
            type = firstLink.mediaType?.toString() ?: "application/xhtml+xml",
            locations = ReadiumLocator.Locations(progression = 0.0),
            text = null,
            title = firstLink.title
        )

        val firstLocator = Locator.Textual(
            bookId = "readio-spike-epub",
            format = BookFormat.EPUB,
            spineId = firstReadiumLocator.href,
            progression = firstReadiumLocator.locations.progression ?: 0.0,
            contextSnippet = firstReadiumLocator.title,
        )

        EpubProbeResult(
            tableOfContents = publication.tableOfContents.map { it.title ?: it.href },
            spineItemCount = publication.readingOrder.size,
            firstSpineHref = firstLink.href,
            firstLocator = firstLocator,
        )
    }

    override suspend fun openPublication(assetPath: String): org.readium.r2.shared.publication.Publication = withContext(Dispatchers.IO) {
        val file = assetCopier.copyToCache(assetPath)

        val httpClient = DefaultHttpClient()
        val assetRetriever = AssetRetriever(context.contentResolver, httpClient)
        val publicationParser = DefaultPublicationParser(context, httpClient, assetRetriever, null)
        val publicationOpener = PublicationOpener(publicationParser)

        val asset = assetRetriever.retrieve(file.toUrl())
            .getOrElse { throw IllegalStateException("Failed to retrieve EPUB asset", it) }

        publicationOpener.open(asset, allowUserInteraction = false)
            .getOrElse {
                asset.close()
                throw IllegalStateException("Failed to open EPUB publication", it)
            }
    }

    override suspend fun toReadiumLocator(locator: Locator.Textual): ReadiumLocator = withContext(Dispatchers.Default) {
        ReadiumLocator(
            href = locator.spineId,
            type = "application/xhtml+xml",
            locations = ReadiumLocator.Locations(progression = locator.progression),
            text = null,
            title = locator.contextSnippet,
        )
    }
}
```

Synchronization rule for this test:
- Do **not** assert immediately after `ActivityScenario.launch()`.
- Do **not** block inside `scenario.onActivity { ... }`.
- The fragment owns the synchronization primitive; the activity only proxies it through `awaitNavigatorSequence(timeoutMs)`.
- Define the stage model explicitly as `NavigatorProofStage.OPENED_PUBLICATION`, `NAVIGATOR_ATTACHED`, `JUMP_REQUESTED`, `LANDED_ON_TARGET`, `RETURN_REQUESTED`, `RETURNED_TO_PRE_JUMP`.
- Mark each stage complete only when this exact event occurs:
  1. `OPENED_PUBLICATION`: `openPublication(...)` finished successfully.
  2. `NAVIGATOR_ATTACHED`: the navigator fragment instance is attached and usable.
  3. `JUMP_REQUESTED`: `navigator.go(targetLocator, animated = false)` was invoked for the evidence target.
  4. `LANDED_ON_TARGET`: `navigator.currentLocator` emitted a locator whose `href` equals the second-resource target href.
  5. `RETURN_REQUESTED`: `navigator.go(preJumpReadiumLocator, animated = false)` was invoked.
  6. `RETURNED_TO_PRE_JUMP`: `navigator.currentLocator` emitted a locator whose `href` equals `preJumpReadiumLocator.href`.
- `awaitNavigatorSequence(timeoutMs)` must throw with the unfinished `NavigatorProofStage` name in the failure message if timeout expires.
- Wait on the **test thread** by calling `scenario.onActivity { activityRef = it }` once, then `activityRef.awaitNavigatorSequence(timeoutMs = 10_000)` outside the callback.

Test shape:

```kotlin
@RunWith(AndroidJUnit4::class)
class EpubNavigatorHostActivityTest {
    @Test
    fun launches_navigator_jumps_to_evidence_anchor_and_returns_to_pre_jump_locator() {
        val context = ApplicationProvider.getApplicationContext<Context>()
        val engine = ReadiumEpubEngine(context)
        val publication = runBlocking { engine.openPublication("fixtures/readio_spike.epub") }
        val firstLink = publication.readingOrder[0]
        val secondLink = publication.readingOrder[1]
        val targetLocator = Locator.Textual(
            bookId = "readio-spike-epub",
            format = BookFormat.EPUB,
            spineId = secondLink.href,
            progression = 0.0,
            contextSnippet = secondLink.title ?: "第2章 夜路"
        )
        val evidenceAnchor = EvidenceAnchor(
            quote = "雨停之后，街口只剩脚步声。",
            locator = targetLocator,
            chapterTitle = "第2章 夜路",
            createdAt = 1_713_654_400_000,
            fallbackAnchor = "chapter2.xhtml#雨停之后，街口只剩脚步声。"
        )

        ActivityScenario.launch<EpubNavigatorHostActivity>(
            EpubNavigatorHostActivity.intent(
                context = context,
                assetPath = "fixtures/readio_spike.epub",
                locator = Locator.Textual(
                    bookId = "readio-spike-epub",
                    format = BookFormat.EPUB,
                    spineId = firstLink.href,
                    progression = 0.0,
                    contextSnippet = firstLink.title ?: "第1章 风起"
                )
            )
        ).use { scenario ->
            var activityRef: EpubNavigatorHostActivity? = null
            scenario.onActivity { activityRef = it }

            val activity = checkNotNull(activityRef)
            activity.awaitNavigatorSequence(timeoutMs = 10_000)

            scenario.onActivity { currentActivity ->
                assertEquals(secondLink.href, evidenceAnchor.locator.spineId)
                assertNotEquals(currentActivity.preJumpReadiumLocator?.href, currentActivity.lastJumpReadiumLocator?.href)
                assertEquals(secondLink.href, currentActivity.lastJumpReadiumLocator?.href)
                assertEquals(firstLink.href, currentActivity.preJumpReadiumLocator?.href)
                assertEquals(firstLink.href, currentActivity.returnedReadiumLocator?.href)
                assertEquals(firstLink.href, currentActivity.latestReadiumLocator?.href)
            }
        }
    }
}
```
